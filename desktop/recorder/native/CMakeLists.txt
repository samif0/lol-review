cmake_minimum_required(VERSION 3.24)
project(revu_recorder VERSION 1.0 LANGUAGES CXX)
include(CTest)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Revu recorder requires Windows x64")
endif()

if(BUILD_TESTING)
  add_executable(revu-encoder-policy-tests encoder-policy-tests.cpp)
  target_compile_features(revu-encoder-policy-tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(revu-encoder-policy-tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME hardware-encoder-policy COMMAND revu-encoder-policy-tests)
  add_executable(revu-frame-stats-tests frame-stats-tests.cpp)
  target_compile_features(revu-frame-stats-tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(revu-frame-stats-tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME recording-frame-stats COMMAND revu-frame-stats-tests)
  add_executable(revu-capture-pacing-tests capture-pacing-tests.cpp)
  target_compile_features(revu-capture-pacing-tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(revu-capture-pacing-tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME wgc-capture-interval COMMAND revu-capture-pacing-tests)
endif()

set(OBS_SOURCE_DIR "" CACHE PATH "Verified pinned OBS source tree")
set(OBS_LIBRARY "" CACHE FILEPATH "Import library from the pinned obs.dll")
set(OBS_CONFIG_DIR "" CACHE PATH "Directory containing matching generated obsconfig.h")
set(OBS_RUNTIME_DIR "" CACHE PATH "Prepared runtime directory beside obs.dll")
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs.h" OR NOT EXISTS "${OBS_LIBRARY}")
  message(FATAL_ERROR "Run the pinned recorder setup script first; OBS_SOURCE_DIR and OBS_LIBRARY are required")
endif()
if(NOT EXISTS "${OBS_CONFIG_DIR}/obsconfig.h")
  message(FATAL_ERROR "OBS_CONFIG_DIR must contain the matching generated obsconfig.h")
endif()

add_executable(revu-recorder recorder.cpp)
target_compile_features(revu-recorder PRIVATE cxx_std_20)
target_compile_definitions(revu-recorder PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(revu-recorder PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_CONFIG_DIR}")
target_link_libraries(revu-recorder PRIVATE "${OBS_LIBRARY}" user32 ole32 dwmapi windowsapp)
if(MSVC)
  target_compile_options(revu-recorder PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(revu-recorder PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
endif()
if(OBS_RUNTIME_DIR)
  add_custom_command(TARGET revu-recorder POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:revu-recorder>" "${OBS_RUNTIME_DIR}/revu-recorder.exe")
endif()
